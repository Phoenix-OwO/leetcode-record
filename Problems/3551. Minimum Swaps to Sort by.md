# 3551. Minimum Swaps to Sort by

[Link](https://leetcode.com/problems/minimum-swaps-to-sort-by-digit-sum/description/)

## Notes

想成環就可以union find 了

```C++
class Solution {
private:
    class UnionFind {
        private:
            vector<int> pa;
            vector<int> rank;
            vector<int> size;
        public:
            explicit UnionFind(int n)
                : pa(n), size(n, 1), rank(n, 1) {
                    iota(pa.begin(), pa.end(), 0);
            }
            int find(int u) {
                if (pa[u] != u) {
                    pa[u] = find(pa[u]);
                }
                return pa[u];
            }
            void unite(int u, int v) {
                int pu = find(u), pv = find(v);
                if (pu == pv) {
                    return ;
                }
                if (rank[pu] < rank[pv]) {
                    pa[pu] = pa[pv];
                    size[pv] += size[pu];
                } else if (rank[pu] > rank[pv]) {
                    pa[pv] = pa[pu];
                    size[pu] += size[pv];
                } else {
                    pa[pu] = pa[pv];
                    size[pv] += size[pu];
                    rank[pv] += 1;
                }
                return ;
            }
            int getSize(int u) {
                return size[u];
            }
    };
public:
    int minSwaps(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> newNum;
        int cnt = 0;

        for (int i = 0; i < n; i++)  {
            newNum.push_back({digSum(nums[i]), nums[i], i});
        }

        sort(newNum.begin(), newNum.end());
        UnionFind uf(n);
        for (int i = 0; i < n; i++) {
            uf.unite(newNum[i][2], i);
        }
        for (int i = 0; i < n; i++) {
            if (uf.find(i) == i) {
                cnt += uf.getSize(i) - 1;
            }
        }
        return cnt;
    }
    int digSum(int num) {
        int ans = 0;
        while (num > 0) {
            ans += num % 10;
            num /= 10;
        }
        return ans;
    }
};
```
