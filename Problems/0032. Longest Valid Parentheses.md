每遇到一個多的右括號，我們就把離他最近的左括號標記成已配對。
整條走完之後我們由左到右看最長連續的已配對之substring 即可。


```C++
class Solution {
public:
    int longestValidParentheses(string s) {
        int n = s.size();
        int cnt = 0, currMax = 0;
        vector<bool> isValid(n, false);
        stack<int> st;
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') 
                st.push(i);
            else {
                if (st.size() != 0) {
                    isValid[st.top()] = true;
                    isValid[i] = true;
                    st.pop();
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (isValid[i] == true) {
                cnt += 1;
            } else {
                currMax = max(currMax, cnt);
                cnt = 0;
            }
        }
        currMax = max(currMax, cnt);
        return currMax;
    }
};
```
