大概就是照做的題目，越來越會寫這種把東西寫在外面的題目（？
總之就是記得當我們要刪掉一個東西的時候把他的time 換成0。

```C++
class BusBooking {
private:
    int currWin = 0;
    int currMid = 0;
    int currTime = 0;
    unordered_map<string, int> time;

public:
    BusBooking(int n) {
        
    }
    
    void toggle(string seat) {
        char row = seat[0];
        string num = seat.substr(1);
        if (time[seat] != 0) {
            if (row == 'A' or row == 'D')
                currWin -= 1;
            else
                currMid -= 1;
            currTime -= time[seat];
            time[seat] = 0;
        }
        else {
            if (row == 'A') {
                currWin += 1;
                if (time['B' + num] != 0) {
                    currTime += 3;
                    time[seat] = 3;
                } else {
                    currTime += 1;
                    time[seat] = 1;
                }
            } else if (row == 'D') {
                currWin += 1;
                if (time['C' + num] != 0) {
                    currTime += 3;
                    time[seat] = 3;
                } else {
                    currTime += 1;
                    time[seat] = 1;
                }
            } else {
                currMid += 1;
                currTime += 1;
                time[seat] = 1;
            }
        }
    }
    
    int getTotalTime() {
        return currTime;
    }
    
    int getMinTime() {
        return currWin + currMid;
    }
};
```
