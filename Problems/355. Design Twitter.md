總之呢，是麻煩的一題，他需要滿多細節的。
比如應該要分人來決定，從他的最新的post 把post 拿出來，所以每個人要有自己的pq 。
如果使用一個PQ 去存所有人的posts 的話會超時。
這題麻煩的點是C++ 容器的使用方法，寫到快中風

```C++
class Twitter {
private:
    int time = 0;
    unordered_map<int, unordered_set<int>> followerList;
    unordered_map<int, priority_queue<vector<int>>> posts;

public:
    Twitter() {

    }
    
    void postTweet(int userId, int tweetId) {
        posts[userId].push({time, tweetId});
        time += 1;
    }
    
    vector<int> getNewsFeed(int userId) {
        int postUser, postId, postTime;
        vector<int> newsFeed;
        vector<vector<int>> tmpPosts;
        priority_queue<vector<int>> currPosts;

        followerList[userId].insert(userId);
        for (const auto& postUser: followerList[userId]) {
            if (!posts[postUser].empty()) {
                postTime = posts[postUser].top()[0];
                postId = posts[postUser].top()[1];
                posts[postUser].pop();
                currPosts.push({postTime, postId, postUser});
                tmpPosts.push_back({postUser, postTime, postId});
            }
        }
        while (!currPosts.empty() and newsFeed.size() < 10) {
            postUser = currPosts.top()[2];
            postId = currPosts.top()[1];
            newsFeed.push_back(postId);
            currPosts.pop();
            if (!posts[postUser].empty()) {
                postTime = posts[postUser].top()[0];
                postId = posts[postUser].top()[1];
                posts[postUser].pop();
                currPosts.push({postTime, postId, postUser});
                tmpPosts.push_back({postUser, postTime, postId});
            }
        }
        
        for (const auto& post: tmpPosts) {
            posts[post[0]].push({post[1], post[2]});
        }
        return newsFeed;
    }
    
    void follow(int followerId, int followeeId) {
        followerList[followerId].insert(followeeId);
    }
    
    void unfollow(int followerId, int followeeId) {
        followerList[followerId].erase(followeeId);
    }
};
```
