# 207. Course Schedule

```C++
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> post(numCourses);
        vector<int> cnt(numCourses);
        deque<int> q;
        int currCourse;

        for (int i = 0; i < prerequisites.size(); i++) {
            post[prerequisites[i][1]].push_back(prerequisites[i][0]);
            cnt[prerequisites[i][0]] += 1;
        }
        for (int i = 0; i < numCourses; i++) {
            if (cnt[i] == 0)
                q.push_back(i);
        }
        while (!q.empty()) {
            currCourse = q.front();
            q.pop_front();
            for (auto nextCourse: post[currCourse]) {
                cnt[nextCourse] -= 1;
                if (cnt[nextCourse] == 0) 
                    q.push_back(nextCourse);
            }
        }
        for (int i = 0; i < numCourses; i++) {
            if (cnt[i] != 0)
                return false;
        }
        return true;

    }
};
```
