```C++
class Solution {
public:
    int sumCounts(vector<int>& nums) {
        int cnt = 0;
        int n = nums.size();
        unordered_set<int> st;

        for (int i = 0; i < n; i++) {
            
            for (int j = i; j < n; j++) {
                st.insert(nums[j]);
                cnt += st.size() * st.size();
            }
            st.clear();
        }
        return cnt;
    }
};
```
