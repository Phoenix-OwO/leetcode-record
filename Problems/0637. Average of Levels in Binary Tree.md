```C++
class Solution {
private:
    vector<long long> nodeSum;
    vector<int> nodeCnt;
public:
    vector<double> averageOfLevels(TreeNode* root) {
        dfs(root, 0);
        vector<double> ans;
        for (int i = 0; i < nodeSum.size(); i++) {
            ans.push_back(double(nodeSum[i])/nodeCnt[i]);
        }
        return ans;
    }
    void dfs(TreeNode* node, int d) {
        if (d >= nodeSum.size()) {
            nodeSum.push_back(0);
            nodeCnt.push_back(0);
        }
        nodeSum[d] += node->val;
        nodeCnt[d] += 1;
        if (node->left) 
            dfs(node->left, d + 1);
        if (node->right)
            dfs(node->right, d + 1);
    }
};
```
