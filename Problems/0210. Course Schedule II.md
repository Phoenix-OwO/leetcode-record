# 210. Course Schedule II

[Link](https://leetcode.com/problems/course-schedule-ii/)

**Topics**: Graph, Topological Sort

## Notes

歡樂topological sort 可能算基本題
就是前一題再加一個array 去存他就好
最後要記得判斷到底可不可以修完這門課

```C++
class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> post(numCourses);
        vector<int> cnt(numCourses);
        vector<int> ans;
        deque<int> q;
        int currCourse;

        for (int i = 0; i < prerequisites.size(); i++) {
            post[prerequisites[i][1]].push_back(prerequisites[i][0]);
            cnt[prerequisites[i][0]] += 1;
        }
        for (int i = 0; i < numCourses; i++) {
            if (cnt[i] == 0) {
                q.push_back(i);
                ans.push_back(i);
            }
        }
        if (!q.empty())
            cout<<q.front()<<endl;
        while (!q.empty()) {
            currCourse = q.front();
            q.pop_front();
            for (auto nextCourse: post[currCourse]) {
                cnt[nextCourse] -= 1;
                if (cnt[nextCourse] == 0) {
                    q.push_back(nextCourse);
                    ans.push_back(nextCourse);
                }
                   
            }
        }
        for (int i = 0; i < numCourses; i++) {
            if (cnt[i] != 0) {
                return {};
            }
        }
        return ans;
    }
};
```
