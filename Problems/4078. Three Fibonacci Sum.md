[link](https://leetcode.com/problems/three-fibonacci-sum/)
照做即可，原本想說要預處理費式數列，後來發現其實不用。


```C++
class Solution {
public:
    bool threeFibonacciSum(int n) {
        int fib0 = 0, fib1 = 1, fib2 = 1;
        int curr = fib0 + fib1 + fib2;
        int tmp;
        if (curr == n) {
            return true;
        }
        while (curr < n) {
            tmp = fib1 + fib2;
            curr += tmp - fib0;
            fib0 = fib1;
            fib1 = fib2;
            fib2 = tmp;
        }
        if (n == curr) {
            return true;
        }
        return false;
    }
};
```
