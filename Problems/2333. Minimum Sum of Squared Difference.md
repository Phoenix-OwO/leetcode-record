# 2333. Minimum Sum of Squared Difference

**Topics**: Binary Search, Greedy, Sorting

## Notes

活用餘數以及尾端加入0 是關鍵，貪心的把所有的數字變小→把選中的變成跟下一個一樣大 如果不夠的話，就善用divmod ，把cnt %k 個變小k%cnt + 1 個 剩下的變小k%cnt

```C++
class Solution {
public:
    long long minSumSquareDiff(vector<int>& nums1, vector<int>& nums2, int k1, int k2) {
        int n = nums1.size();
        int k = k1 + k2;
        long long curr = 0;
        vector<long long> diff(n);
        long long ans = 0;
        for (int i = 0; i < n; i++) {
            diff[i] = abs(nums1[i] - nums2[i]);
            curr += diff[i];
        }
        if (curr <= k) {
            return 0;
        }
        sort(diff.rbegin(), diff.rend());
        diff.push_back(0);
        curr = diff[0];

        for (int i = 0; i <= n; i++) {
            if ((curr - diff[i]) * i <= k) {
                k -= (curr - diff[i]) * i;
                curr = diff[i];
            } else {
                long long delta = k / i;
                ans += (curr - delta) * (curr - delta) * (i - k % i);
                ans += (curr - delta - 1) * (curr - delta - 1) * (k % i);
                for (int j = i; j < n; j++) {
                    ans += diff[j] * diff[j];
                }
                return ans;
            }
        }
        return ans;
    }
};
```
